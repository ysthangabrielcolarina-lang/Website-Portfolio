package com.pokedex.pokedex.model;

import jakarta.persistence.*; // imports jpa annotations

@Entity // turn Java class into a database table
public class Pokemon {

    @Id // identifies each row in a database table.
    @GeneratedValue(strategy = GenerationType.IDENTITY) // matic assign id
    private Long id;
    private String name;
    private String type;
    private String description;
    private Double height;
    private Double weight;
    private String imageUrl;

    public Pokemon() {
    }

    public Pokemon(String name, String type, String description,
                   Double height, Double weight, String imageUrl) {
        this.name = name;
        this.type = type;
        this.description = description;
        this.height = height;
        this.weight = weight;
        this.imageUrl = imageUrl;
    }

    public Long getId() { // since naka private, need ng getters/setters para mamodify
        return id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getType() {
        return type;
    }

    public void setType(String type) {
        this.type = type;
    }

    public String getDescription() {
        return description;
    }

    public void setDescription(String description) {
        this.description = description;
    }

    public Double getHeight() {
        return height;
    }

    public void setHeight(Double height) {
        this.height = height;
    }

    public Double getWeight() {
        return weight;
    }

    public void setWeight(Double weight) {
        this.weight = weight;
    }

    public String getImageUrl() {
        return imageUrl;
    }

    public void setImageUrl(String imageUrl) {
        this.imageUrl = imageUrl;
    }
}
