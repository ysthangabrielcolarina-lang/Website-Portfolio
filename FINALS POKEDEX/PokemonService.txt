    // READ ALL
    public List<Pokemon> getAllPokemon() {
        return repository.findAll();
    }

    // CREATE
    public Pokemon addPokemon(Pokemon pokemon) {

        // Check if name is empty
        if (pokemon.getName() == null || pokemon.getName().trim().isEmpty()) {
            throw new RuntimeException("Name is required!");
        }

        // Check if type is empty
        if (pokemon.getType() == null || pokemon.getType().trim().isEmpty()) {
            throw new RuntimeException("Type is required!");
        }

        // Check for negative or zero height
        if (pokemon.getHeight() == null || pokemon.getHeight() <= 0) {
            throw new RuntimeException("Height must be greater than 0!");
        }

        // Check for negative or zero weight
        if (pokemon.getWeight() == null || pokemon.getWeight() <= 0) {
            throw new RuntimeException("Weight must be greater than 0!");
        }

        // Check for duplicate Pokemon name (case-insensitive)
        boolean duplicateName = repository.findAll().stream()
                .anyMatch(p -> p.getName().equalsIgnoreCase(pokemon.getName()));

        if (duplicateName) {
            throw new RuntimeException("Pokemon with this name already exists!");
        }

        return repository.save(pokemon);
    }

    // UPDATE
    public Pokemon updatePokemon(Long id, Pokemon updatedPokemon) {
        Pokemon existingPokemon = repository.findById(id)
                .orElseThrow(() -> new RuntimeException("Pokemon not found"));

        if (updatedPokemon.getHeight() > 0) {
            existingPokemon.setHeight(updatedPokemon.getHeight());
        }

        if (updatedPokemon.getWeight() > 0) {
            existingPokemon.setWeight(updatedPokemon.getWeight());
        }

        if (updatedPokemon.getDescription() != null && !updatedPokemon.getDescription().isEmpty()) {
            existingPokemon.setDescription(updatedPokemon.getDescription());
        }

        return repository.save(existingPokemon);
    }

    // DELETE
    public void deletePokemon(Long id) {
        repository.deleteById(id);
    }
